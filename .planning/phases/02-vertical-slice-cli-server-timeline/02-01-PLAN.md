```ts
import { basename } from "node:path";
import { Buffer } from "node:buffer";
import { Correlator, EventStore, projectRow, bandFor, type EventRow, type LatencyBand } from "@ahp-viewer/core";
import type { HostAdapter, LogHandle } from "@ahp-viewer/host-node"; // adjust import to actual export
import { LineSplitter, normalize, parseLine } from "@ahp-viewer/parser";
import { type Direction, makeParseErrorEvent } from "@ahp-viewer/shared";
import type { Status } from "@ahp-viewer/core";

export interface LogMeta {
  readonly filename: string;   // basename only — never absolute path
  readonly sizeBytes: number;
  readonly startedAt: number;
}

export type SsePayload =
  | { kind: "snapshot-begin"; meta: LogMeta; total: number }
  | { kind: "snapshot-chunk"; rows: EventRow[]; from: number }
  | { kind: "snapshot-end" }
  | { kind: "append"; rows: EventRow[]; from: number }
  | { kind: "patch"; updates: Array<{ idx: number; status: Status; latencyMs: number | null; latencyBand: LatencyBand | null }> }
  | { kind: "ping" }
  | { kind: "bye" }
  | { kind: "error"; code: string; message: string };

export type Listener = (payload: SsePayload) => void;

export interface AppStateOptions {
  readonly host: HostAdapter;
  readonly file: string;
  readonly directionInference?: (raw: unknown) => Direction; // injected by CLI in 02-05
  readonly flushIntervalMs?: number;   // default 1000
  readonly unmatchedTimeoutMs?: number; // default 30000
}

export interface AppState {
  readonly meta: LogMeta;
  snapshot(): { meta: LogMeta; rows: EventRow[] };
  subscribe(listener: Listener): () => void;
  dispose(): Promise<void>;
}

export async function createAppState(opts: AppStateOptions): Promise<AppState> {
  const handle: LogHandle = await opts.host.openLog(opts.file);
  const store = new EventStore();
  const correlator = new Correlator(store);
  const splitter = new LineSplitter();
  const decoder = new TextDecoder("utf-8");
  const inferDir = opts.directionInference ?? (() => "c2s" as Direction);

  const rows: EventRow[] = [];
  const listeners = new Set<Listener>();
  const meta: LogMeta = {
    filename: basename(handle.path),
    sizeBytes: handle.size ?? 0,
    startedAt: Date.now(),
  };

  let seq = 0;
  let byteOffset = 0;

  // Build/refresh row[idx] from store + correlator state.
  function buildRow(idx: number): EventRow {
    const ev = store.at(idx)!;
    const status = correlator.statusOf(idx);
    const latency = correlator.latencyOf(idx);
    return projectRow(ev, idx, status, latency);
  }

  // After append: track which earlier requests' rows changed because a response patched them.
  let lastSize = 0;
  store.subscribe((range) => {
    const newRows: EventRow[] = [];
    for (let i = range.from; i < range.to; i++) {
      const row = buildRow(i);
      rows[i] = row;
      newRows.push(row);
    }
    // Detect retro patches: any earlier request whose status flipped from 'pending' to 'ok'/'error'/'unmatched'/'orphan'.
    const updates: Array<{ idx: number; status: Status; latencyMs: number | null; latencyBand: LatencyBand | null }> = [];
    for (let i = 0; i < range.from; i++) {
      const newStatus = correlator.statusOf(i);
      const newLatency = correlator.latencyOf(i);
      if (rows[i]!.status !== newStatus || rows[i]!.latencyMs !== newLatency) {
        const prev = rows[i]!;
        rows[i] = { ...prev, status: newStatus, latencyMs: newLatency, latencyBand: bandFor(newLatency) };
        updates.push({ idx: i, status: newStatus, latencyMs: newLatency, latencyBand: bandFor(newLatency) });
      }
    }
    if (newRows.length > 0) emit({ kind: "append", rows: newRows, from: range.from });
    if (updates.length > 0) emit({ kind: "patch", updates });
    lastSize = store.size();
  });

  function emit(payload: SsePayload): void {
    for (const l of listeners) {
      try { l(payload); } catch { /* never let a bad listener block ingest */ }
    }
  }

  // Ingest loop driven by host.watchLog.
  const watcher = opts.host.watchLog(handle, (chunk: Uint8Array) => {
    const text = decoder.decode(chunk, { stream: true });
    for (const line of splitter.push(text)) {
      const byteLength = Buffer.byteLength(line, "utf8");
      const ts = Date.now();
      const parsed = parseLine(line, byteOffset, byteLength);
      const dir: Direction = parsed.error ? "c2s" : inferDir(parsed.raw);
      const m = { seq, ts, tsRaw: String(ts), dir, byteOffset, byteLength };
      const ev = parsed.error ? makeParseErrorEvent(m, parsed.error.reason, parsed.text) : normalize(parsed.raw, m);
      store.append(ev);
      seq += 1;
      byteOffset += byteLength + 1;
    }
  });

  // Flush ticker — drives transition to 'unmatched'.
  const flushIntervalMs = opts.flushIntervalMs ?? 1000;
  const unmatchedTimeoutMs = opts.unmatchedTimeoutMs ?? 30_000;
  const flushTimer = setInterval(() => {
    correlator.flush(Date.now(), unmatchedTimeoutMs);
    // Replay the same diff scan synthetically by triggering a no-op range probe.
    const updates: Array<{ idx: number; status: Status; latencyMs: number | null; latencyBand: LatencyBand | null }> = [];
    for (let i = 0; i < store.size(); i++) {
      const cur = correlator.statusOf(i);
      const curLat = correlator.latencyOf(i);
      const prev = rows[i];
      if (prev && (prev.status !== cur || prev.latencyMs !== curLat)) {
        rows[i] = { ...prev, status: cur, latencyMs: curLat, latencyBand: bandFor(curLat) };
        updates.push({ idx: i, status: cur, latencyMs: curLat, latencyBand: bandFor(curLat) });
      }
    }
    if (updates.length > 0) emit({ kind: "patch", updates });
  }, flushIntervalMs);

  let disposed = false;
  return {
    meta,
    snapshot() { return { meta, rows: rows.slice() }; },
    subscribe(l) { listeners.add(l); return () => listeners.delete(l); },
    async dispose() {
      if (disposed) return;
      disposed = true;
      clearInterval(flushTimer);
      try { watcher.dispose(); } catch { /* ignore */ }
      correlator.dispose();
      listeners.clear();
    },
  };
}
```

```ts
import type { MiddlewareHandler } from "hono";

const ALLOWED_HOST_RE = /^(?:127\.0\.0\.1|localhost)(?::\d{1,5})?$/i;

export const hostGuardMiddleware: MiddlewareHandler = async (c, next) => {
  const host = c.req.header("host") ?? "";
  if (!ALLOWED_HOST_RE.test(host)) {
    return c.text("Misdirected request", 421);
  }
  await next();
};
```

```ts
import type { Hono } from "hono";
import { streamSSE } from "hono/streaming";
import type { AppState, SsePayload } from "./app-state.js";

const SNAPSHOT_CHUNK = 2000;

export function registerLogRoutes(app: Hono, appState: AppState): void {
  app.get("/api/log/meta", (c) => c.json(appState.meta));

  app.get("/api/log/stream", (c) =>
    streamSSE(c, async (stream) => {
      const snap = appState.snapshot();
      await stream.writeSSE({ event: "snapshot-begin", data: JSON.stringify({ meta: snap.meta, total: snap.rows.length }) });
      for (let i = 0; i < snap.rows.length; i += SNAPSHOT_CHUNK) {
        await stream.writeSSE({
          event: "snapshot-chunk",
          data: JSON.stringify({ rows: snap.rows.slice(i, i + SNAPSHOT_CHUNK), from: i }),
        });
        await stream.sleep(0);
      }
      await stream.writeSSE({ event: "snapshot-end", data: "{}" });

      const off = appState.subscribe(async (msg: SsePayload) => {
        try {
          await stream.writeSSE({ event: msg.kind, data: JSON.stringify(msg) });
        } catch { /* aborted */ }
      });
      c.req.raw.signal.addEventListener("abort", off);

      while (!c.req.raw.signal.aborted) {
        await stream.sleep(20_000);
        if (c.req.raw.signal.aborted) break;
        await stream.writeSSE({ event: "ping", data: "{}" });
      }
    }));
}
```

```ts
import { type ServerType, serve } from "@hono/node-server";
import { Hono } from "hono";
import type { AppState } from "./app-state.js";
import { cspMiddleware } from "./csp.js";
import { hostGuardMiddleware } from "./host-guard.js";
import { registerLogRoutes } from "./sse-routes.js";

const HOSTNAME = "127.0.0.1" as const;

export interface LogServerOptions {
  readonly appState: AppState;
  readonly port: number;       // 0 = ephemeral
  readonly version: string;
  readonly uiDistDir?: string; // optional; production static UI mount
}

export interface LogServerHandle {
  readonly url: string;
  readonly port: number;
  readonly server: ServerType;
  close(): Promise<void>;
  /** Send `event: bye` to all current SSE clients. Best-effort. */
  sayGoodbye(): void;
}

export function startLogServer(opts: LogServerOptions): Promise<LogServerHandle> {
  const app = new Hono();
  app.use("*", hostGuardMiddleware);
  app.use("*", cspMiddleware);
  app.get("/health", (c) => c.json({ status: "ok", version: opts.version }));
  registerLogRoutes(app, opts.appState);

  // Static UI serving — only if dist dir provided. Plan 02-06 wires this up.
  // For Phase 2 we keep it optional; the integration test does not exercise it.

  return new Promise<LogServerHandle>((resolve, reject) => {
    const server = serve(
      { fetch: app.fetch, hostname: HOSTNAME, port: opts.port },
      (info) => {
        const port = info.port;
        resolve({
          url: `http://${HOSTNAME}:${port}`,
          port,
          server,
          sayGoodbye() {
            // The bye signal flows through AppState listeners; the route handler
            // emits whatever AppState sends. CLI calls AppState.dispose which
            // could itself emit a final 'bye' payload — see plan 02-05.
          },
          close: () => new Promise<void>((res, rej) => server.close((err) => (err ? rej(err) : res()))),
        });
      },
    );
    server.on("error", reject);
  });
}
```
