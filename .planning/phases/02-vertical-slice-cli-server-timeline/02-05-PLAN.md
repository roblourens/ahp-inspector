```ts
import type { Direction } from "@ahp-viewer/shared";

/**
 * Classify wire direction from a JSON-RPC envelope, assuming this CLI is
 * viewing a CLIENT-SIDE log (e.g. VS Code's Copilot AHP log). For server-side
 * logs, callers will pass a different inference function in a future flag.
 *
 * Rules (in order):
 *  1. Non-object → 'c2s' (defensive default)
 *  2. method === 'action' or 'notification' → 's2c' (server-originated)
 *  3. has 'result' or 'error' (no method) → 's2c' (incoming response)
 *  4. has 'method' and 'id' → 'c2s' (outbound request)
 *  5. has 'method' (notification, no id) → 'c2s' (client notification)
 *  6. fallback → 'c2s'
 */
export function classifyDirection(raw: unknown): Direction {
  if (raw === null || typeof raw !== "object") return "c2s";
  const r = raw as { method?: unknown; id?: unknown; result?: unknown; error?: unknown };
  if (r.method === "action" || r.method === "notification") return "s2c";
  if ("result" in r || "error" in r) return "s2c";
  if (typeof r.method === "string") return "c2s";
  return "c2s";
}
```

```ts
#!/usr/bin/env node
import { readFileSync, statSync } from "node:fs";
import { dirname, resolve as resolvePath, join } from "node:path";
import { fileURLToPath } from "node:url";
import { Command } from "commander";
import open from "open";
import { NodeHostAdapter } from "@ahp-viewer/host-node";
import { createAppState, startLogServer } from "@ahp-viewer/server";
import { classifyDirection } from "./direction.js";

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);

function loadVersion(): string {
  const candidates = [join(__dirname, "..", "package.json"), join(__dirname, "package.json")];
  for (const p of candidates) {
    try { return (JSON.parse(readFileSync(p, "utf8")) as { version?: string }).version ?? "0.0.0"; }
    catch { /* try next */ }
  }
  return "0.0.0";
}
const VERSION = loadVersion();

function fail(msg: string, code = 1): never {
  process.stderr.write(msg.endsWith("\n") ? msg : `${msg}\n`);
  process.exit(code);
}

function parsePort(value: string): number {
  if (!/^-?\d+$/.test(value)) fail(`Error: invalid --port value: ${value}. Use 0–65535.`);
  const n = Number(value);
  if (!Number.isInteger(n) || n < 0 || n > 65535) {
    fail(`Error: invalid --port value: ${value}. Use 0–65535.`);
  }
  return n;
}

const program = new Command()
  .name("ahp-viewer")
  .version(VERSION)
  .argument("[file]", "AHP JSONL log file path")
  .option("--port <n>", "local server port (0 = ephemeral)", "5173")
  .option("--no-open", "do not auto-open the default browser")
  .action(async (file: string | undefined, opts: { port: string; open: boolean }) => {
    if (!file) {
      fail(`Error: log file not found: <missing>\nUsage: ahp-viewer <path-to-log.jsonl>`);
    }
    const absPath = resolvePath(file);
    let stat;
    try { stat = statSync(absPath); }
    catch (err) {
      const e = err as NodeJS.ErrnoException;
      if (e.code === "ENOENT") fail(`Error: log file not found: ${absPath}\nUsage: ahp-viewer <path-to-log.jsonl>`);
      fail(`Error: cannot read ${absPath}: ${e.message}\nCheck file permissions.`);
    }
    if (!stat!.isFile()) fail(`Error: log file not found: ${absPath}\nUsage: ahp-viewer <path-to-log.jsonl>`);

    const port = parsePort(opts.port);

    const host = new NodeHostAdapter();
    const appState = await createAppState({ host, file: absPath, directionInference: classifyDirection });

    let serverHandle;
    try {
      serverHandle = await startLogServer({ appState, port, version: VERSION });
    } catch (err) {
      const e = err as NodeJS.ErrnoException;
      if (e.code === "EADDRINUSE") {
        await appState.dispose();
        fail(`Error: port ${port} is in use. Try: ahp-viewer --port ${port + 1} ${absPath}`);
      }
      await appState.dispose();
      fail(`Error: failed to start server: ${e.message}`);
    }

    const url = `http://127.0.0.1:${serverHandle.port}`;
    process.stdout.write(`AHP Log Viewer running at ${url}\nOpening browser…\nWatching ${absPath}\n`);

    if (opts.open !== false) {
      try { await open(url, { wait: false }); }
      catch { process.stdout.write(`(could not auto-open; visit ${url})\n`); }
    }

    let shuttingDown = false;
    const shutdown = async () => {
      if (shuttingDown) return;
      shuttingDown = true;
      try { serverHandle.sayGoodbye(); } catch { /* ignore */ }
      try { await appState.dispose(); } catch { /* ignore */ }
      try { await serverHandle.close(); } catch { /* ignore */ }
      process.exit(0);
    };
    process.on("SIGINT", () => { void shutdown(); });
    process.on("SIGTERM", () => { void shutdown(); });
  });

program.parseAsync().catch((err) => {
  process.stderr.write(`Error: ${(err as Error).message}\n`);
  process.exit(1);
});
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| user CLI argv → server start | File path + port number cross from untrusted shell into local file/network ops |
| CLI → default browser | URL passed to `open()` becomes a navigation target |

## STRIDE Threat Register

| ID | Category | Component | Disposition | Mitigation |
|----|----------|-----------|-------------|-----------|
| T-02-05a | Tampering | --port argv | mitigate | `parsePort` rejects non-integers and out-of-range values with explicit error copy + exit 1; tests cover `70000` and `-5` |
| T-02-05b | Tampering / SSRF | browser open URL | mitigate | `open(url)` is invoked exclusively with the constructed `http://127.0.0.1:${serverHandle.port}` string; no user-supplied URL flows in; `--no-open` flag opts out; honors `BROWSER=none` via the `open` package |
| T-02-05c | Information Disclosure | error messages | accept | Error messages echo absolute path because the user owns the path (UI-SPEC §10 copy mandates it for the `Watching` line); never echoed to the browser/SSE — that path stays terminal-only |
| T-02-05d | DoS | repeat invocation | mitigate | Port-in-use error gives an explicit retry hint and exits cleanly (no orphan processes); SIGINT/SIGTERM disposes AppState + server within `shutdown()` |
| T-02-05e | Spoofing | server bind address | mitigate | Server binding is delegated to `startLogServer` which hard-codes `127.0.0.1`; CLI never passes a hostname |
