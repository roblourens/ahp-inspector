```ts
function isEditableTarget(el: EventTarget | null): boolean {
  if (!(el instanceof HTMLElement)) return false;
  const tag = el.tagName;
  if (tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT") return true;
  if (el.isContentEditable) return true;       // [contenteditable="true"]
  return false;
}

function onKeyDown(ev: React.KeyboardEvent<HTMLElement>): void {
  if (ev.key !== " " && ev.code !== "Space") return;
  if (isEditableTarget(ev.target)) return;     // SearchInput, ManualOpenInput, etc.
  ev.preventDefault();                          // prevent page scroll
  const s = useAppStore.getState();
  s.setLivePaused(!s.livePaused);
}
```

1. Reads `logKey`, `rows.length`, and a snapshot version counter from the store.
