```ts
export interface PerLogPrefs {
  v: 1;
  searchQuery: string;
  filters: FilterState;
  grouping: GroupingMode;
  groupCollapsed: string[];
  selectedIdx: number | null;
  detailWidth: number;
  livePaused: boolean;
}
```
Old records use inclusion semantics for categorical arrays other than `method`; they must not be interpreted as hidden arrays.
