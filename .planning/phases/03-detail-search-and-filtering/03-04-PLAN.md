**`packages/ui/src/transport/http-client.ts`** — MUST NOT import `node:`, `fs`,
`path`, `hono`, or `@ahp-viewer/server` (boundary test enforces this):

```ts
// Simple LRU-16 cache keyed by event idx.
const MAX_CACHE = 16;
const cache = new Map<number, DetailResponse>();
const cacheOrder: number[] = [];

function cacheGet(idx: number): DetailResponse | undefined {
  return cache.get(idx);
}
function cacheSet(idx: number, data: DetailResponse): void {
  if (cache.has(idx)) {
    cacheOrder.splice(cacheOrder.indexOf(idx), 1);
    cacheOrder.push(idx);
    cache.set(idx, data);
    return;
  }
  if (cacheOrder.length >= MAX_CACHE) {
    const evict = cacheOrder.shift()!;
    cache.delete(evict);
  }
  cacheOrder.push(idx);
  cache.set(idx, data);
}

export async function fetchEvent(
  idx: number,
  signal?: AbortSignal,
): Promise<DetailResponse | null> {
  const cached = cacheGet(idx);
  if (cached) return cached;
  const resp = await fetch(`/api/log/event/${idx}`, { signal });
  if (resp.status === 404) return null;
  if (!resp.ok) throw new Error(`Failed to load event: ${resp.status}`);
  const data: DetailResponse = await resp.json();
  cacheSet(idx, data);
  return data;
}

export type { DetailResponse };
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| GET /api/log/event/:idx response → browser DOM | Full AhpEvent including raw payload — must not be rendered via dangerouslySetInnerHTML |
| navigator.clipboard.writeText | Explicit user action only; contents may include tokens/prompts (disclosed in PrivacyCaption) |
| react-json-view-lite render tree | Library renders text values — verified no eval in Plan 03-00 |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-04-01 | Tampering (XSS) | RawJsonView / PrettyJsonView | mitigate | `<pre>{string}</pre>` — React auto-escapes; `react-json-view-lite` text-only; NO dangerouslySetInnerHTML anywhere; contract test in DetailPanel.test.tsx with `<script>` payload |
| T-03-04-02 | DoS (UI freeze) | PrettyJsonView with huge payload | mitigate | 256KB client cap: if `JSON.stringify(data).length > 262144`, render TruncationBanner instead of tree; raw view still renders (single string, bounded) |
| T-03-04-03 | Info Disclosure (clipboard) | CopyMenu | mitigate | Copy only on explicit user click; PrivacyCaption discloses that raw payload (tokens, prompts, paths) is included |
| T-03-04-04 | DoS (listener leak) | DetailPanel fetchEvent effect | mitigate | AbortController created per selectedIdx; previous controller aborted on new selectedIdx change in useEffect cleanup |
| T-03-04-05 | Info Disclosure (path leak) | DetailResponse event field | accept | AhpEvent has no path field; any paths in raw payload are user-logged data, not server internals; server sanitizes LogMeta separately (Plan 03-01) |
</threat_model>
