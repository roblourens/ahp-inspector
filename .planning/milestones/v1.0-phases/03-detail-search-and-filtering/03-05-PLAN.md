**`packages/ui/src/transport/search-client.ts`** (NO node imports):
```ts
export interface SearchResult {
  matches: number[];
  total: number;
  truncated: boolean;
}

export async function searchEvents(
  q: string,
  signal?: AbortSignal,
): Promise<SearchResult> {
  const params = new URLSearchParams({ q, limit: "5000" });
  const resp = await fetch(`/api/log/search?${params.toString()}`, { signal });
  if (!resp.ok) throw new Error(`Search failed: ${resp.status}`);
  return resp.json() as Promise<SearchResult>;
}
```

**`packages/ui/src/components/filters/useSearch.ts`**:
```ts
import { useEffect, useRef } from "react";
import { useAppStore } from "../../state/store.js";
import { searchEvents } from "../../transport/search-client.js";

const DEBOUNCE_MS = 150;

export function useSearch(): void {
  const query = useAppStore((s) => s.searchQuery);
  const setMatches = useAppStore((s) => s.setSearchMatches);
  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
  const abortRef = useRef<AbortController | null>(null);

  useEffect(() => {
    if (timerRef.current) clearTimeout(timerRef.current);
    if (!query) {
      // Empty query — clear matches immediately (T-03-08: cancel any in-flight)
      abortRef.current?.abort();
      setMatches(null);
      return;
    }
    timerRef.current = setTimeout(() => {
      abortRef.current?.abort();
      const ctrl = new AbortController();
      abortRef.current = ctrl;
      searchEvents(query, ctrl.signal)
        .then((r) => setMatches(r.matches))
        .catch((err: unknown) => {
          if ((err as { name?: string })?.name === "AbortError") return;
          // Non-abort error: surface via store or console
          console.error("[useSearch]", err);
        });
    }, DEBOUNCE_MS);
    return () => {
      if (timerRef.current) clearTimeout(timerRef.current);
    };
  }, [query, setMatches]);
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| searchQuery (user text) → GET /api/log/search | User-typed text crosses to server; server enforces 256-char cap and substring-only (Plan 03-01) |
| AbortController → fetch | Client-side only; aborts must not leave zombie listeners |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-05-01 | DoS (listener leak) | useSearch AbortController | mitigate | Each query change creates a new AbortController; previous one is aborted; useEffect cleanup also aborts on unmount |
| T-03-05-02 | DoS (UI block) | TimelineList re-render with 50k grouped items | mitigate | `useGroupedItems` is memoized via `useMemo([rows, filteredRowIdxs, grouping])`; TanStack Virtual renders only visible slice; perf covered by selectors.perf.test.ts |
| T-03-05-03 | Tampering (XSS) | highlightMatches in EventRow | mitigate | Uses React `<mark>` element with text children — auto-escaped; searchQuery never interpreted as HTML |
| T-03-05-04 | Info Disclosure | isAuthFailure glyph | accept | Auth failure is visible metadata about the user's own log; not a secret |
</threat_model>
