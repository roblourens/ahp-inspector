**1. Create `packages/ui/src/transport/sse-client.ts`:**

```ts
import type { EventRow, LatencyBand } from "@ahp-viewer/core";
import type { Status } from "@ahp-viewer/core";
import { useAppStore } from "../state/store.js";

export interface ConnectionHandle { close(): void; }

interface SnapshotBegin { meta: { filename: string; sizeBytes: number; startedAt: number }; total: number; }
interface SnapshotChunk { rows: EventRow[]; from: number; }
interface AppendMsg     { rows: EventRow[]; from: number; }
interface PatchMsg      { updates: Array<{ idx: number; status: Status; latencyMs: number | null; latencyBand: LatencyBand | null }>; }

export interface ConnectOpts {
  url?: string;                                  // default '/api/log/stream'
  EventSourceCtor?: typeof EventSource;          // injected for tests
}

export function connectLogStream(opts: ConnectOpts = {}): ConnectionHandle {
  const url = opts.url ?? "/api/log/stream";
  const Ctor = opts.EventSourceCtor ?? globalThis.EventSource;
  const store = useAppStore.getState();
  store.setConnection("connecting");

  const es = new Ctor(url);
  let snapshotRows: EventRow[] = [];
  let graceful = false;

  es.addEventListener("snapshot-begin", (ev) => {
    const data = JSON.parse((ev as MessageEvent).data) as SnapshotBegin;
    snapshotRows = [];
    useAppStore.getState().setMeta({ filename: data.meta.filename, eventCount: 0, sessionCount: 0 });
  });
  es.addEventListener("snapshot-chunk", (ev) => {
    const data = JSON.parse((ev as MessageEvent).data) as SnapshotChunk;
    snapshotRows = snapshotRows.concat(data.rows);
  });
  es.addEventListener("snapshot-end", () => {
    useAppStore.getState().setRows(snapshotRows);
    snapshotRows = [];
    useAppStore.getState().setConnection("connected");
  });
  es.addEventListener("append", (ev) => {
    const data = JSON.parse((ev as MessageEvent).data) as AppendMsg;
    useAppStore.getState().appendRows(data.rows, data.from);
  });
  es.addEventListener("patch", (ev) => {
    const data = JSON.parse((ev as MessageEvent).data) as PatchMsg;
    useAppStore.getState().applyPatch(data.updates);
  });
  es.addEventListener("ping", () => { /* no-op heartbeat */ });
  es.addEventListener("bye", () => {
    graceful = true;
    es.close();
    useAppStore.getState().setConnection("disconnected");
  });
  es.onerror = () => {
    if (graceful) return;
    if (es.readyState === 2 /* CLOSED */) useAppStore.getState().setConnection("disconnected");
    else useAppStore.getState().setConnection("connecting");
  };

  return { close: () => { graceful = true; es.close(); } };
}
```

```tsx
import { useEffect } from "react";
import { AppShell } from "./components/shell/AppShell.js";
import { useAppStore } from "./state/store.js";
import { ServerNotRunningState } from "./components/states/ServerNotRunningState.js";
import { connectLogStream } from "./transport/sse-client.js";

export function App(): JSX.Element {
  const connection = useAppStore((s) => s.connection);

  useEffect(() => {
    // Probe meta first; if 503 / network-down, show no-server state.
    let cancelled = false;
    fetch("/api/log/meta").then(
      (r) => {
        if (cancelled) return;
        if (!r.ok) { useAppStore.getState().setConnection("no-server"); return; }
        const handle = connectLogStream();
        // store handle on window for the disconnected-banner reconnect flow
        (window as unknown as { __ahpStream?: { close(): void } }).__ahpStream = handle;
      },
      () => { if (!cancelled) useAppStore.getState().setConnection("no-server"); }
    );
    return () => { cancelled = true; };
  }, []);

  if (connection === "no-server") return <ServerNotRunningState />;
  return <AppShell />;
}
```
